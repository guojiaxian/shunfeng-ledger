/* OCR 前的图片压缩：最长边压到 1536px 并重编码为 JPEG(q80)。
 * 目的：① 减小 base64 请求体（大图直传上传慢，部分 API 拒收超过 10MB 的请求）；
 *      ② detail:'high' 下视觉 token 按 ~512px 分块编码，降采样能明显缩短模型编码耗时。
 * 金额/地址等关键文字在 1536px 长边下仍清晰可读；解码或编码失败时原样返回，不阻塞识别。 */
function compressForOcr(dataUrl, maxEdge = 1536, quality = 0.8){
  return new Promise(resolve=>{
    const img = new Image();
    img.onload = ()=>{
      try{
        const scale = Math.min(1, maxEdge / Math.max(img.width, img.height));
        const w = Math.max(1, Math.round(img.width * scale));
        const h = Math.max(1, Math.round(img.height * scale));
        const canvas = document.createElement('canvas');
        canvas.width = w; canvas.height = h;
        const ctx = canvas.getContext('2d');
        ctx.fillStyle = '#fff';               // JPEG 无透明通道，铺白底
        ctx.fillRect(0, 0, w, h);
        ctx.imageSmoothingEnabled = true;
        if('imageSmoothingQuality' in ctx) ctx.imageSmoothingQuality = 'high';
        ctx.drawImage(img, 0, 0, w, h);
        const out = canvas.toDataURL('image/jpeg', quality);
        // 重编码后反而更大（原图已是小体积 JPEG）时退回原图
        resolve(out.length < dataUrl.length ? out : dataUrl);
      }catch(e){
        resolve(dataUrl);
      }
    };
    img.onerror = ()=> resolve(dataUrl);
    img.src = dataUrl;
  });
}

function haversineKm(lat1, lng1, lat2, lng2){
  const rad = Math.PI / 180, R = 6371;
  const dLat = (lat2 - lat1) * rad, dLng = (lng2 - lng1) * rad;
  const a = Math.sin(dLat / 2) * Math.sin(dLat / 2) +
            Math.cos(lat1 * rad) * Math.cos(lat2 * rad) * Math.sin(dLng / 2) * Math.sin(dLng / 2);
  return 2 * R * Math.asin(Math.sqrt(a));
}

 * 而 XMLHttpRequest 在 allowUniversalAccessFromFileURLs=true 下可绕过 CORS。
 * 因此统一用 XHR 实现，连接测试与 OCR 识别共用。
 * 返回 { ok, status, statusText, text }，超时抛 Error('timeout')。
 */
function llmHttpPost(url, headers, body, timeoutMs){
  return new Promise((resolve, reject)=>{
    const xhr = new XMLHttpRequest();
    xhr.open('POST', url, true);
    // 显式设置 Content-Type，确保服务端按 JSON 解析
    xhr.setRequestHeader('Content-Type', 'application/json');
    if(headers){
      for(const k in headers){
        // XHR 不允许设置某些受保护头（如 Accept-Charset），跳过 Content-Type（已设置）
        if(k.toLowerCase() === 'content-type') continue;
        try{ xhr.setRequestHeader(k, headers[k]); }catch(e){ /* 受保护头忽略 */ }
      }
    }
    xhr.timeout = timeoutMs || 30000;
    xhr.ontimeout = ()=> reject(new Error('连接超时（请检查网络或 API 地址）'));
    xhr.onerror = ()=>{
      // XHR 在网络层失败（DNS、TLS、CORS 被拦截）时触发，无状态码
      reject(new Error('网络请求失败（status=' + xhr.status + '）—— 可能是 API 地址错误、网络不通或被拦截'));
    };
    xhr.onload = ()=>{
      resolve({
        ok: xhr.status >= 200 && xhr.status < 300,
        status: xhr.status,
        statusText: xhr.statusText,
        text: xhr.responseText
      });
    };
    xhr.onabort = ()=> reject(new Error('请求已取消'));
    xhr.send(body);
  });
}
